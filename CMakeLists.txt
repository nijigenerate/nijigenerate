cmake_minimum_required(VERSION 3.24)
if(NOT DEFINED CMAKE_MSVC_RUNTIME_LIBRARY)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
endif()
project(nijigenerate_native LANGUAGES CXX)
include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/NativeCompilePolicy.cmake")

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)

# No configure step downloads dependencies or invokes Git.
option(NIJIGENERATE_RUNTIME_PROBE "Build the nicxlive/SDL3 integration probe" OFF)
option(NIJIGENERATE_BUILD_EDITOR "Build the actual native editor executable" OFF)
option(NIJIGENERATE_EDITOR_HAVE_MCP "Enable the editor's existing MCP service feature" ON)
option(NIJIGENERATE_EDITOR_IN_NIGHTLY "Enable the editor's nightly-build notice" OFF)
option(NIJIGENERATE_EDITOR_IN_LITE "Use the editor's lite CRT/subsystem policy" OFF)
set(NIJIGENERATE_NICXLIVE_SOURCE_DIR "" CACHE PATH "Local nicxlive source tree")
set(NIJIGENERATE_D_COMPILER "" CACHE FILEPATH "Optional LDC executable for D reference tests")
set(NIJIGENERATE_PYTHON "" CACHE FILEPATH "Optional Python executable for differential tests")
set(NIJIGENERATE_INMATH_SOURCE_DIR "" CACHE PATH "Local D inmath source for numeric reference tests")
set(NIJIGENERATE_SERIALIZER_SOURCE "" CACHE FILEPATH "Local nijilive serialize.d for reference tests")
set(NIJIGENERATE_PUPPET_SOURCE "" CACHE FILEPATH "Local nijilive puppet.d for metadata reference tests")
set(NIJIGENERATE_FGHJ_SOURCE_DIR "" CACHE PATH "Local D fghj source for serialization reference tests")
set(NIJIGENERATE_MIR_ALGORITHM_SOURCE_DIR "" CACHE PATH "Local D mir-algorithm source")
set(NIJIGENERATE_MIR_CORE_SOURCE_DIR "" CACHE PATH "Local D mir-core source")

add_library(nijigenerate_native_options INTERFACE)
target_compile_features(nijigenerate_native_options INTERFACE cxx_std_20)
if(MSVC)
    target_compile_options(nijigenerate_native_options INTERFACE /W4 /utf-8 /fp:strict)
else()
    target_compile_options(nijigenerate_native_options INTERFACE -Wall -Wextra -Wpedantic -ffp-contract=off)
endif()
add_subdirectory(native)
